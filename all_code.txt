private fun startProgram() {

        "Two pointers Sum",
        "Sliding Window",
        "Exit"
    )

    while (true) {
        println("==================================================")
        println("Algorithms Menu")
        println("==================================================\n")
        options.forEachIndexed { index, option ->
            println("${index + 1} - $option")
        }

        print("\nType the desired option: ")
        val optionString = readlnOrNull()?.trim()
        when (val option = optionString?.toIntOrNull()) {
            null -> {
                println("Please, enter a valid option.")
            }
            3 -> {
                println("\nExiting program. Farewell!")
                break
            }
            else -> {
                chooseAlgorithm(option)
            }
        }
    }
}

    when (option) {
        1 -> TwoPointerSum.startTwoPointersProgram()
        2 -> SlidingWindow.startSlidingWindowProgram()
    }

package algorithms

import model.SlidingWindowDTO
import readNumberList
import readInt

object SlidingWindow {
    internal fun startSlidingWindowProgram() {
        val numberList = readNumberList()

        val window = readInt("Provide the desired number quantity: ")

        validateWindow(numberList, window)
    }

    internal fun validateWindow(numberList: List<Float>, window: Int) : SlidingWindowDTO? {
        if (numberList.size < window) {
            println("The number quantity is bigger than the number quantity in the list.")
            return null
        } else {
            val slidingWindow = slidingWindowSum(numberList, window)

            printResult(slidingWindow)
            return slidingWindow
        }
    }

    internal fun slidingWindowSum(numbers: List<Float>, window: Int): SlidingWindowDTO {

        require(window in 1..numbers.size) {
            "Window must be between 1 and ${numbers.size}"
        }

        var firstIndex = 0
        var lastIndex = window - 1
        var sum: Float
        var maxSum: Float
        var average: Float
        var slidingWindow: SlidingWindowDTO
        val sumList = numbers.subList(firstIndex, window)
        sum = sumList.sum()
        maxSum = sum
        slidingWindow = SlidingWindowDTO(
            window = window,
            sum = maxSum,
            average = maxSum / window,
            numberList = numbers,
            indexes = 0..lastIndex
        )

        while (lastIndex < numbers.size - 1) {
            val outgoingNumber = numbers[firstIndex]
            firstIndex++
            lastIndex = firstIndex + (window - 1)
            val currentSum = sum - outgoingNumber + numbers[lastIndex]
            sum = currentSum

            if (maxSum < currentSum) {
                maxSum = currentSum
                average = currentSum / window
                slidingWindow = SlidingWindowDTO(
                    window = window,
                    sum = maxSum,
                    average = average,
                    numberList = numbers,
                    indexes = firstIndex..lastIndex
                )
            }
        }

        return slidingWindow
    }

    private fun printResult(slidingWindowDTO: SlidingWindowDTO?) {
        if (slidingWindowDTO == null) {
            println("No sliding-window found")
        } else {
            println("Using the list ${slidingWindowDTO.numberList} and getting ${slidingWindowDTO.window} numbers from this list" +
                    "on index ${slidingWindowDTO.indexes}, " +
                    "we get a sum result ${slidingWindowDTO.sum} with ${slidingWindowDTO.average} as average.")
        }
    }
}

package algorithms

import getIndexLabel
import readNumberList
import readFloat

object TwoPointerSum {
    internal fun startTwoPointersProgram() {
        val numberList = readNumberList()

        val targetNumber = readFloat("Provide the desired sum target: ")


        val sortedList = numberList.sorted()
        val result = twoPointersSum(sortedList, targetNumber)

        printResult(result, sortedList, targetNumber)
    }

    internal fun twoPointersSum(numberList: List<Float>, target: Float): Pair<Int, Int>? {
        var leftPointIndex = 0
        var rightPointIndex = numberList.size - 1
        while (leftPointIndex < rightPointIndex) {
            val sum = numberList[leftPointIndex] + numberList[rightPointIndex]

            if (sum == target) {
                return Pair(leftPointIndex, rightPointIndex)
            } else if (sum < target) {
                leftPointIndex++
            } else {
                rightPointIndex--
            }
        }

        return null
    }

    private fun printResult(result: Pair<Int, Int>?, numberList: List<Float>, target: Float) {
        if (result == null) {
            println("It is not possible to achieve your target using the provided number list.\n")
        } else {

            println(
                "To achieve your target, we sum the ${getIndexLabel(result.first + 1)} " +
                        "number and the ${getIndexLabel(result.second + 1)} number.\n"
            )
            println("${numberList[result.first]} + ${numberList[result.second]} = $target\n")
        }
    }
}

        assertNull(validWindow)
    }

    @Test
    fun `Should not allow a window greater than the number list size on the DTO construction`() {
        assertThrows<IllegalArgumentException> {
            SlidingWindowDTO(
                window = greaterWindow,
                numberList = numberList,
                sum = 0f,
                average = 0f,
                indexes = 0..1,
            )
        }
    }

    @Test
    fun `Should reject a window size of zero`() {
        assertThrows<IllegalArgumentException> {
            SlidingWindowDTO(
                window = 0,
                numberList = numberList,
                sum = 0f,
                average = 0f,
                indexes = 0..1,
            )
        }
    }

    @Test
    fun `Should not allow a window greater than the number list size`() {
        assertThrows<IllegalArgumentException> {
            slidingWindowSum(numberList, greaterWindow)
        }

    fun `Should find maximum average when the best window is at the end`() {

        val duplicateTarget = 6f
        val negativeTarget = -1f

        val negativeNumberList = listOf(-5f, -2f, 1f, 4f)
        val duplicateNumberList = listOf(2f, 3f, 3f, 7f)

        val result = twoPointersSum(numberList, target)
        val sum = result?.let {
            numberList[it.first] + numberList[it.second]
        }

        val duplicateResult = twoPointersSum(duplicateNumberList, duplicateTarget)
        val duplicateSum = duplicateResult?.let {
            duplicateNumberList[it.first] + duplicateNumberList[it.second]
        }

        val negativeResult = twoPointersSum(negativeNumberList, negativeTarget)
        val negativeSum = negativeResult?.let {
            negativeNumberList[it.first] + negativeNumberList[it.second]
        }

        assertEquals(target, sum)
        assertEquals(duplicateTarget, duplicateSum)
        assertEquals(negativeTarget, negativeSum)
    }

    @Test
    fun `Should not find two pointers sum result`() {
        val target = 22f
        val emptyList = emptyList<Float>()

        val emptySum = emptyResult?.let {
            emptyList[it.first] + emptyList[it.second]

fun getIndexLabel(index: Int): String {
    return when (index) {
        1 -> "first"
        2 -> "second"
        3 -> "third"
        else -> "${index}th"
    }
}

fun validateNumberList(input: String?): List<Float>? {
    val items = input?.split(",")
    val numberList = mutableListOf<Float>()

    if (items == null) {
        println("Please, provide a list with valid numbers.")
    } else if (items.any { it.toFloatOrNull() == null }) {
        println("All elements must be a number")
    } else if (items.size < 4) {
        println("Please enter at least 4 numbers")
    } else {
        items.forEach { numberList.add(it.trim().toFloat()) }
        return numberList
    }

    return null
}

fun validateTarget(input: String?): Float? {
    val targetNumber = input?.toFloatOrNull()

    if (targetNumber == null) {
        println("Please enter a valid number.")
        return null
    } else {
        return targetNumber
    }
}
